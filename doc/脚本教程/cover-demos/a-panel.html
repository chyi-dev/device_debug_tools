认帧头、取数值
function on_receive(frame)
  if not frame:starts_with(0xA5, 0x5A, 0x03) then return end
  local t = frame:u16(4) / 10
  local h = frame:u16(6) / 10
  log(string.format("温度 %.1f℃ 湿度 %.1f%%", t, h))
end
样例 A5 5A 03 00 F6 02 09 03温度 24.6℃　湿度 52.1%
